CREATE TYPE cuisine_cart_types AS ENUM ('checkbox', 'radio');
SELECT enum_range(NULL::cuisine_cart_types);

CREATE TABLE "cuisine_cart" (
  "id" integer PRIMARY KEY GENERATED ALWAYS AS IDENTITY (sequence name "cuisine_cart_id_seq"),
  "cuisine_id" smallint NOT NULL,
  "cuisine_cart_type" cuisine_cart_types NOT NULL,
  "group" varchar(25) NOT NULL,
  "name" varchar(50) NOT NULL,
  "price" smallint DEFAULT 0 NOT NULL,
  "flag" boolean DEFAULT true NOT NULL
);

ALTER TABLE "cuisine_cart"
ADD CONSTRAINT "cuisine_cart_fk_cuisine_id" FOREIGN KEY ("cuisine_id") REFERENCES "cuisines" ("id");
